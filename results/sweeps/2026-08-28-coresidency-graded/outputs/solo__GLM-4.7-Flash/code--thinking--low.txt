from typing import List

def parse_ranges(s: str) -> List[int]:
    """
    Parse a string containing ranges separated by commas.

    Whitespace around numbers and commas is ignored. A single number is treated
    as a range of length one. A range in the form "start-end" expands to all
    integers from start to end inclusive. If start > end, a ValueError is raised.
    Empty or whitespace-only strings return an empty list.

    Examples:
    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1-3,5,7-9 ')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError: Reversed range
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('5')
    [5]
    >>> parse_ranges('1-2,4-6')
    [1, 2, 4, 5, 6]
    """
    if not s.strip():
        return []

    result = []
    for part in s.split(','):
        part = part.strip()
        if not part:
            continue  # skip empty parts (e.g., from consecutive commas)

        if '-' in part:
            # Split only on the first hyphen to allow negative numbers if needed
            left_str, right_str = part.split('-', 1)
            left = int(left_str.strip())
            right = int(right_str.strip())
            if left > right:
                raise ValueError("Reversed range")
            result.extend(range(left, right + 1))
        else:
            num = int(part.strip())
            result.append(num)

    return result
