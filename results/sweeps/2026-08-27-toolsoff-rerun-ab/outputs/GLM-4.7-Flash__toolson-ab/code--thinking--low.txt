def parse_ranges(s: str) -> list[int]:
    """
    Parse a string of comma-separated ranges into a list of integers.
    
    Whitespace around numbers and commas is ignored. A reversed range (e.g., '9-7')
    raises a ValueError. Empty or whitespace-only input returns an empty list.
    
    Args:
        s: A string containing comma-separated ranges like '1-3,5,7-9'
        
    Returns:
        A list of integers in the expanded form
        
    Raises:
        ValueError: If a range is reversed (e.g., '9-7')
        
    Doctests:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]
        
        >>> parse_ranges(' 1 - 3 , 5 , 7 - 9 ')
        [1, 2, 3, 5, 7, 8, 9]
        
        >>> parse_ranges('10-12,15,18-20')
        [10, 11, 12, 15, 18, 19, 20]
        
        >>> parse_ranges('5')
        [5]
        
        >>> parse_ranges('')
        []
        
        >>> parse_ranges('   ')
        []
    """
    if not s or s.strip() == '':
        return []
    
    result = []
    for token in s.split(','):
        token = token.strip()
        if not token:
            continue
        if '-' in token:
            start_str, end_str = token.split('-')
            start = int(start_str.strip())
            end = int(end_str.strip())
            if end < start:
                raise ValueError(f"Invalid range: {start_str}-{end_str}")
            result.extend(range(start, end + 1))
        else:
            result.append(int(token))
    return result
