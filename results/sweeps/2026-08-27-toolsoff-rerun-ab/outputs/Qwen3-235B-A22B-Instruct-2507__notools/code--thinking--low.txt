def parse_ranges(s: str) -> list[int]:
    """
    Parse a string of ranges and numbers into a list of integers.
    
    The input string can contain individual numbers and ranges separated by commas.
    A range is specified as 'start-end' where start <= end.
    Whitespace around numbers and commas is ignored.
    
    Args:
        s: A string containing numbers and ranges separated by commas
        
    Returns:
        A list of integers in ascending order, with ranges expanded
        
    Raises:
        ValueError: If a range is reversed (e.g., '9-7')
        
    Examples:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]
        
        >>> parse_ranges('  1 - 3 , 5 , 7 - 9  ')
        [1, 2, 3, 5, 7, 8, 9]
        
        >>> parse_ranges('')
        []
        
        >>> parse_ranges('  ')
        []
        
        >>> parse_ranges('1,3-5,7')
        [1, 3, 4, 5, 7]
        
        >>> parse_ranges('5-3')
        Traceback (most recent call last):
        ValueError: Invalid range: 5-3 (start must be <= end)
        
        >>> parse_ranges('1-,3')
        Traceback (most recent call last):
        ValueError: Invalid range or number: 1-
    """
    if not s or s.isspace():
        return []
    
    result = []
    
    # Split by comma and strip whitespace
    parts = [part.strip() for part in s.split(',') if part.strip()]
    
    for part in parts:
        if '-' in part:
            # It's a range
            range_parts = part.split('-')
            if len(range_parts) != 2:
                raise ValueError(f"Invalid range or number: {part}")
            
            try:
                start = int(range_parts[0])
                end = int(range_parts[1])
            except ValueError:
                raise ValueError(f"Invalid range or number: {part}")
                
            if start > end:
                raise ValueError(f"Invalid range: {part} (start must be <= end)")
                
            # Add all numbers in the range [start, end]
            result.extend(range(start, end + 1))
        else:
            # It's a single number
            try:
                result.append(int(part))
            except ValueError:
                raise ValueError(f"Invalid number: {part}")
    
    return result
