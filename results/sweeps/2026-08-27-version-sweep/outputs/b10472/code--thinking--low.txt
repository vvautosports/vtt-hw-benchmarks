```python
def parse_ranges(s: str) -> list[int]:
    """
    Parse a string containing numeric ranges into a list of integers.

    Examples:
    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('1, 3-5, 7')
    [1, 3, 4, 5, 7]
    >>> parse_ranges('1 - 3, 5, 7 - 9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('1-3')
    [1, 2, 3]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError: Reversed range detected: 9-7
    >>> parse_ranges('10-12, 5')
    [10, 11, 12, 5]
    >>> parse_ranges('0-5')
    [0, 1, 2, 3, 4, 5]
    """
    # Strip whitespace from the input
    s = s.strip()

    # Handle empty input
    if not s:
        return []

    # Split by comma and strip whitespace from each part
    parts = [p.strip() for p in s.split(',')]

    result = []

    for part in parts:
        if '-' in part:
            # Parse range
            start_str, end_str = [x.strip() for x in part.split('-')]

            try:
                start = int(start_str)
                end = int(end_str)
            except ValueError as e:
                raise ValueError(f"Invalid number in range '{part}'") from e

            # Check for reversed range
            if start > end:
                raise ValueError(f"Reversed range detected: {part}")

            # Add all values in the range (inclusive)
            result.extend(range(start, end + 1))
        else:
            # Single number
            try:
                num = int(part)
            except ValueError as e:
                raise ValueError(f"Invalid number '{part}'") from e
            result.append(num)

    return result
```

**Key features:**
- ✅ Parses ranges like `'1-3,5,7-9'` into `[1, 2, 3, 5, 7, 8, 9]`
- ✅ Ignores whitespace around numbers and commas
- ✅ Raises `ValueError` for reversed ranges like `'9-7'`
- ✅ Returns `[]` for empty or whitespace-only input
- ✅ Ranges are inclusive (both ends included)
