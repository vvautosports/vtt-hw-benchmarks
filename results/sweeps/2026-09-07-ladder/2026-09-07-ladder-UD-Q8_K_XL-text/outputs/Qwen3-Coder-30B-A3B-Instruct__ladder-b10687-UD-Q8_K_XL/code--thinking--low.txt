def parse_ranges(s: str) -> list[int]:
    """
    Parse a string of ranges into a list of integers.
    
    Args:
        s: A string containing ranges separated by commas, e.g., '1-3,5,7-9'
        
    Returns:
        A list of integers representing all numbers in the ranges
        
    Raises:
        ValueError: If a range is reversed (like '9-7')
        
    Examples:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]
        >>> parse_ranges('  1 - 3 , 5 , 7 - 9  ')
        [1, 2, 3, 5, 7, 8, 9]
        >>> parse_ranges('10-12,15')
        [10, 11, 12, 15]
        >>> parse_ranges('')
        []
        >>> parse_ranges('  ')
        []
        >>> parse_ranges('5-3')
        Traceback (most recent call last):
        ...
        ValueError: Reversed range: 5-3
    """
    # Handle empty or whitespace-only input
    if not s or not s.strip():
        return []
    
    result = []
    
    # Split by comma and process each part
    for part in s.split(','):
        part = part.strip()
        if not part:  # Skip empty parts
            continue
            
        # Check if it's a range (contains '-')
        if '-' in part:
            try:
                start_str, end_str = part.split('-')
                start = int(start_str.strip())
                end = int(end_str.strip())
                
                # Check for reversed range
                if start > end:
                    raise ValueError(f"Reversed range: {start}-{end}")
                
                # Add range to result
                result.extend(range(start, end + 1))
            except ValueError:
                # Re-raise ValueError for invalid range syntax
                raise ValueError(f"Invalid range: {part}")
        else:
            # Single number
            result.append(int(part.strip()))
    
    return result
