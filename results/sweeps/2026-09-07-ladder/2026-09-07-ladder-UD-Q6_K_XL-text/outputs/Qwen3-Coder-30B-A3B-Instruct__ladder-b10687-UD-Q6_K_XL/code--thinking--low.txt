def parse_ranges(s: str) -> list[int]:
    """
    Parse a string of ranges into a list of integers.
    
    Args:
        s: A string containing ranges separated by commas, e.g., '1-3,5,7-9'
        
    Returns:
        A list of integers representing all numbers in the ranges
        
    Raises:
        ValueError: If a range is reversed (e.g., '9-7')
        
    Examples:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]
        >>> parse_ranges('  1 - 3 , 5 , 7 - 9  ')
        [1, 2, 3, 5, 7, 8, 9]
        >>> parse_ranges('10-12')
        [10, 11, 12]
        >>> parse_ranges('42')
        [42]
        >>> parse_ranges('')
        []
        >>> parse_ranges('9-7')
        Traceback (most recent call last):
        ...
        ValueError: Reversed range: 9-7
    """
    # Handle empty or whitespace-only input
    if not s or not s.strip():
        return []
    
    result = []
    
    # Split by commas and process each part
    for part in s.split(','):
        part = part.strip()
        if not part:  # Skip empty parts (shouldn't happen with valid input)
            continue
            
        # Check for range syntax
        if '-' in part:
            range_parts = part.split('-')
            if len(range_parts) != 2:
                raise ValueError(f"Invalid range format: {part}")
            
            start_str, end_str = range_parts
            start = int(start_str.strip())
            end = int(end_str.strip())
            
            # Check for reversed range
            if start > end:
                raise ValueError(f"Reversed range: {start}-{end}")
            
            # Add all numbers in the range
            result.extend(range(start, end + 1))
        else:
            # Single number
            result.append(int(part.strip()))
    
    return result
