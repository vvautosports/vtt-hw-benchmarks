def parse_ranges(s: str) -> list:
    """
    Parses a string containing range specifications and returns a list of integers
    covering all ranges. Each range is specified as 'start-end' or just 'value'.
    
    TL;DR: Extracts integers from ranges like "1-3,5,7-9" into [1,2,3,5,7,8,9].
    """
    result = []
    for part in s.split(','):
        if '-' in part:
            start, end = map(int, part.split('-'))
            result.extend(range(start, end + 1))
        else:
            result.append(int(part))
    return result
