def parse_ranges(s: str) -> list[int]:
    """
    Parse a string of ranges and individual numbers into a list of integers.
    
    Args:
        s (str): A string containing ranges (e.g., '1-3') and individual numbers 
                 (e.g., '5') separated by commas
        
    Returns:
        list[int]: A sorted list of integers from the ranges and numbers
        
    Raises:
        ValueError: If a range has start > end
        
    Examples:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]
        
        >>> parse_ranges('  1 - 3 , 5 , 7 - 9  ')
        [1, 2, 3, 5, 7, 8, 9]
        
        >>> parse_ranges('')
        []
        
        >>> parse_ranges('  ')
        []
        
        >>> parse_ranges('10-5')
        Traceback (most recent call last):
        ...
        ValueError: Invalid range: 10-5 (start cannot be greater than end)
    """
    # Handle empty or whitespace-only input
    if not s or not s.strip():
        return []
    
    result = []
    
    # Split by comma and process each part
    for part in s.split(','):
        part = part.strip()
        if not part:  # Skip empty parts
            continue
            
        # Check if it's a range (contains '-')
        if '-' in part:
            start_str, end_str = part.split('-')
            start = int(start_str.strip())
            end = int(end_str.strip())
            
            if start > end:
                raise ValueError(f"Invalid range: {start}-{end} (start cannot be greater than end)")
            
            # Add all numbers in the range (inclusive)
            result.extend(range(start, end + 1))
        else:
            # Single number
            result.append(int(part.strip()))
    
    return sorted(result)
