import re

_PART_RE = re.compile(r"^(-?\d+)\s*-\s*(-?\d+)$|^(-?\d+)$")


def parse_ranges(s: str) -> list[int]:
    """
    Parse a comma-separated list of integers and inclusive ranges.

    >>> parse_ranges("1-3,5,7-9")
    [1, 2, 3, 5, 7, 8, 9]

    Whitespace around numbers and commas is ignored.

    >>> parse_ranges("  1 - 3 ,  5  ,7- 9 ")
    [1, 2, 3, 5, 7, 8, 9]

    Empty or whitespace-only input yields an empty list.

    >>> parse_ranges("")
    []
    >>> parse_ranges("  \t  ")
    []

    A reversed range raises ValueError.

    >>> parse_ranges("1-3,9-7")
    Traceback (most recent call last):
        ...
    ValueError: reversed range: '9-7'

    Malformed input raises ValueError too.

    >>> parse_ranges("1,abc,3")
    Traceback (most recent call last):
        ...
    ValueError: invalid range element: 'abc'
    """
    numbers: list[int] = []
    for part in s.split(","):
        part = part.strip()
        if not part:
            if s.strip():
                raise ValueError("empty range element")
            continue
        match = _PART_RE.match(part)
        if match is None:
            raise ValueError(f"invalid range element: {part!r}")
        if match.group(3) is not None:          # single number
            numbers.append(int(match.group(3)))
        else:                                    # start-end range
            start, stop = int(match.group(1)), int(match.group(2))
            if start > stop:
                raise ValueError(f"reversed range: {part!r}")
            numbers.extend(range(start, stop + 1))
    return numbers

Notes on design choices:
- **Splitting:** commas separate elements; each element is matched against a regex that accepts either `a-b` or a single `n`, so whitespace is tolerated anywhere around the numbers, comma, and dash.
- **Negatives:** handled correctly (`"-2--1"` → `[-2, -1]`) because the regex greedily takes an optional leading `-` for each number before requiring the range dash.
- **Empty input:** `""` and whitespace-only strings return `[]`; a stray empty element like `"1,,2"` or `","` is treated as malformed and raises `ValueError`.
