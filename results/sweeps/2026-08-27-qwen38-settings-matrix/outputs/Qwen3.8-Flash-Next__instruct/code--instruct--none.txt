```python
def parse_ranges(s: str) -> list[int]:
    """Parse a range string like '1-3,5,7-9' into a flat list of ints.

    Rules:
      * Whitespace around numbers and commas is ignored.
      * A single number expands to itself.
      * An 'a-b' range expands to every integer from a to b inclusive.
      * A reversed range (a > b) raises ValueError.
      * Empty or whitespace-only input returns [].

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 - 3 , 5 , 7-9 ')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('42')
    [42]
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError: reversed range: 9-7
    >>> parse_ranges('1-3,9-7')
    Traceback (most recent call last):
        ...
    ValueError: reversed range: 9-7
    >>> parse_ranges('1,,2')
    Traceback (most recent call last):
        ...
    ValueError: invalid range item: ''
    """
    result: list[int] = []
    if not s.strip():
        return result

    for part in s.split(','):
        part = part.strip()
        if not part:
            raise ValueError(f"invalid range item: {part!r}")
        if '-' in part[1:] or part.startswith('-'):
            # split on the LAST dash only, so negatives stay unambiguous
            left, _, right = part.rpartition('-')
            try:
                start, end = int(left), int(right)
            except ValueError:
                raise ValueError(f"invalid range item: {part!r}") from None
            if start > end:
                raise ValueError(f"reversed range: {part}")
            result.extend(range(start, end + 1))
        else:
            try:
                result.append(int(part))
            except ValueError:
                raise ValueError(f"invalid range item: {part!r}") from None
    return result
```

Notes on the design:

- **Whitespace**: each comma-separated part is `.strip()`ed, and `int()` itself tolerates surrounding spaces inside a range endpoint (e.g. `' 1 - 3 '`), so no manual space-stripping of the whole string is needed beyond the empty check.
- **Empty input**: handled up front with `s.strip()`, returning `[]` for both `''` and whitespace-only strings, rather than falling through to the "invalid item" error.
- **Reversed ranges**: detected by comparing parsed endpoints, raising `ValueError` with the offending text (e.g. `reversed range: 9-7`) so callers can see which part failed, even in a mixed string like `'1-3,9-7'`.
- **Malformed items** (`'1,,2'`, `'abc'`, `'1-x'`) also raise `ValueError` with a consistent message, chained via `from None` for a clean traceback.
