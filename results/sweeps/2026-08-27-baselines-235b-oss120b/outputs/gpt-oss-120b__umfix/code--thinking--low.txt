def parse_ranges(s: str) -> list[int]:
    """
    Parse a comma‑separated list of numbers and ranges into a flat list of integers.

    A *range* is expressed as ``start-end`` where ``start`` and ``end`` are
    non‑negative integers.  The range is inclusive and must be in ascending order
    (``start`` ≤ ``end``); otherwise a :class:`ValueError` is raised.

    Whitespace around numbers, commas and dashes is ignored.  An empty string or a
    string that contains only whitespace yields an empty list.

    Parameters
    ----------
    s: str
        The input string, e.g. ``"1-3, 5, 7-9"``.

    Returns
    -------
    list[int]
        A sorted list of the individual integers represented by the input.

    Raises
    ------
    ValueError
        If any range is reversed (``start`` > ``end``) or contains a non‑integer
        token.

    Doctests
    --------
    >>> parse_ranges("")                 # empty input
    []
    >>> parse_ranges("   ")              # whitespace only
    []
    >>> parse_ranges("1-3,5,7-9")        # normal usage
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(" 2 , 4-4 , 6 ")    # single numbers and a degenerate range
    [2, 4, 6]
    >>> parse_ranges("10-8")             # reversed range → error
    Traceback (most recent call last):
        ...
    ValueError: reversed range '10-8' (start > end)
    >>> parse_ranges("3-5, a, 7")        # non‑integer token → error
    Traceback (most recent call last):
        ...
    ValueError: invalid integer token 'a'
    """
    # Strip the whole string first; if nothing remains, return an empty list.
    s = s.strip()
    if not s:
        return []

    result: list[int] = []
    # Split on commas – commas themselves are not part of the numbers.
    for part in s.split(","):
        part = part.strip()
        if not part:
            # Skip empty parts that could arise from trailing commas etc.
            continue

        if "-" in part:
            # It's a range; split on the first dash to allow numbers like "1-3-5" to raise.
            start_str, end_str = map(str.strip, part.split("-", 1))
            try:
                start = int(start_str)
                end = int(end_str)
            except ValueError:
                raise ValueError(f"invalid integer token '{start_str if not start_str.isdigit() else end_str}'")
            if start > end:
                raise ValueError(f"reversed range '{part}' (start > end)")
            result.extend(range(start, end + 1))
        else:
            # Single number.
            try:
                num = int(part)
            except ValueError:
                raise ValueError(f"invalid integer token '{part}'")
            result.append(num)

    # The specification does not require deduplication or sorting beyond the
    # natural order produced by the algorithm, but sorting makes the function
    # more predictable when inputs like "5,1-3" are given.
    return sorted(result)
