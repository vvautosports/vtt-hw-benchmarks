```python
def parse_ranges(s: str) -> list[int]:
    """
    Parse a range string like ``'1-3,5,7-9'`` into a flat list of integers.

    Supported syntax
    ----------------
    * ``<start>-<end>`` – an inclusive range (``start`` must be ≤ ``end``).
    * ``<number>`` – a single value.
    * Items are separated by commas.
    * Whitespace around numbers, commas, and the dash is ignored.

    Errors
    ------
    * ``ValueError`` is raised if a range is given in reversed order
      (e.g. ``'9-7'``).

    Examples
    --------
    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 - 3 , 5 , 7 - 9 ')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
    ValueError

    """
    # Strip surrounding whitespace; if nothing left, return [].
    s = s.strip()
    if not s:
        return []

    result: list[int] = []
    # Split on commas – also strip whitespace around each token.
    parts = [part.strip() for part in s.split(',')]

    for part in parts:
        if not part:                      # skip empty tokens like "1,,3"
            continue
        if '-' in part:
            # Split only on the first dash to avoid issues with stray dashes.
            start_str, end_str = part.split('-', 1)
            start = int(start_str.strip())
            end = int(end_str.strip())
            if start > end:
                raise ValueError("reversed range")
            # Extend inclusive range.
            result.extend(range(start, end + 1))
        else:
            result.append(int(part))

    return result
```
