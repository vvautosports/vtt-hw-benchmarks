def parse_ranges(s: str) -> list[int]:
    """
    Parse a range string like ``'1-3,5,7-9'`` into a list of integers.

    Whitespace around numbers and commas is ignored.
    A reversed range (e.g. ``'9-7'``) raises ``ValueError``.
    Empty or whitespace‑only input returns ``[]``.

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('  10  ')
    [10]
    >>> parse_ranges('1 - 3 , 5 , 7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError: reversed range '9-7'
    >>> parse_ranges('   ')
    []
    """
    s = s.strip()
    if not s:
        return []

    result: list[int] = []
    for part in s.split(','):
        part = part.strip()
        if not part:
            continue                     # skip empty bits from stray commas
        if '-' in part:
            start_str, end_str = part.split('-', 1)
            start = int(start_str.strip())
            end   = int(end_str.strip())
            if start > end:
                raise ValueError(f"reversed range '{part}'")
            # start <= end  →  include every integer from start to end inclusive
            if start == end:
                result.append(start)
            else:
                result.extend(range(start, end + 1))
        else:
            result.append(int(part))
    return result
