def parse_ranges(s: str) -> list[int]:
    """
    Parse a range string like '1-3,5,7-9' into a list of integers.

    Whitespace around numbers and commas is ignored.
    A reversed range like '9-7' raises ValueError.
    Empty or whitespace-only input returns [].

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 , 3-5 ')
    [1, 3, 4, 5]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
    ValueError: reversed range '9-7'
    """
    s = s.strip()
    if not s:
        return []
    parts = s.split(',')
    result = []
    for part in parts:
        part = part.strip()
        if not part:
            continue
        if '-' in part:
            start_str, end_str = part.split('-', 1)
            start = int(start_str.strip())
            end = int(end_str.strip())
            if start > end:
                raise ValueError(f"reversed range '{part}'")
            result.extend(range(start, end + 1))
        else:
            result.append(int(part))
    return result
