def parse_ranges(s: str) -> list[int]:
    """
    Parse a string containing comma‑separated ranges into a list of ints.

    Whitespace around numbers and commas is ignored.
    A range ``a-b`` expands to all integers from ``a`` to ``b`` inclusive.
    If ``a > b`` the function raises :class:`ValueError`.
    Empty or whitespace‑only input returns an empty list.

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 10 , 11-12 ')
    [10, 11, 12]
    >>> parse_ranges('5-5')
    [5]
    >>> parse_ranges('  ')
    []
    >>> parse_ranges('8-6')
    Traceback (most recent call last):
        ...
    ValueError: Invalid range: 8-6 (start > end)

    """
    s = s.strip()
    if not s:
        return []

    result: list[int] = []
    for token in s.split(','):
        token = token.strip()
        if not token:
            continue
        if '-' in token:
            start_str, end_str = token.split('-', 1)
            start = int(start_str.strip())
            end = int(end_str.strip())
            if start > end:
                raise ValueError(f"Invalid range: {token} (start > end)")
            result.extend(range(start, end + 1))
        else:
            result.append(int(token.strip()))
    return result
