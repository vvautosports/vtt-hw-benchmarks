```python
def parse_ranges(s: str) -> list[int]:
    """
    Parses a string of comma-separated ranges and single numbers into a list of integers.

    The string can contain single numbers (e.g., '5') or ranges (e.g., '1-3').
    Whitespace around numbers and commas is ignored.
    A range where the start is greater than the end (e.g., '9-7') raises a ValueError.
    An empty or whitespace-only string returns an empty list.

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 - 2, 4 , 6 - 6 ')
    [1, 2, 4, 6]
    >>> parse_ranges('  ')
    []
    >>> parse_ranges('10')
    [10]
    >>> parse_ranges('5-1')
    Traceback (most recent call last):
        ...
    ValueError: Reversed range: 5-1
    """
    s = s.strip()
    if not s:
        return []

    result = []
    # Split by comma to get individual segments
    segments = s.split(',')

    for segment in segments:
        segment = segment.strip()
        if not segment:
            continue

        if '-' in segment:
            # Split the segment into start and end parts
            parts = segment.split('-')
            if len(parts) != 2:
                raise ValueError(f"Invalid range format: '{segment}'")
            
            try:
                start = int(parts[0].strip())
                end = int(parts[1].strip())
            except ValueError:
                raise ValueError(f"Non-integer value found in range: '{segment}'")

            if start > end:
                raise ValueError(f"Reversed range: {start}-{end}")
            
            # range(start, end + 1) includes the 'end' value
            result.extend(range(start, end + 1))
        else:
            # Handle single number
            try:
                result.append(int(segment))
            except ValueError:
                raise ValueError(f"Invalid number format: '{segment}'")

    return result
```
