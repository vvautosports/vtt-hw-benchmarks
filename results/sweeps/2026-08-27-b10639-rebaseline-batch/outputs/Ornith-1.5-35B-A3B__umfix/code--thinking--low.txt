```python
def parse_ranges(s: str) -> list[int]:
    """Parse a comma-separated string of integers and ranges into a flat list.

    Whitespace around numbers and commas is ignored. A single number ``n``
    yields ``[n]``; a range ``a-b`` yields ``[a, a+1, ..., b]``. The range
    endpoints must be in ascending order, otherwise a :class:`ValueError` is
    raised. Empty or whitespace-only input yields ``[]``.

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('  10 , 12 , 14-15 ')
    [10, 12, 14, 15]
    >>> parse_ranges('42')
    [42]
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError: reversed range '9-7'

    :param s: the range string to parse
    :raises ValueError: if a range's start is greater than its end, or if the
        input contains malformed tokens
    :return: the flattened list of integers
    """
    numbers: list[int] = []

    tokens = s.split(',')
    # '' and '   ' all split into a single empty token, so this short-circuits
    # to [] before we ever try to interpret any component.
    if not any(token.strip() for token in tokens):
        return []

    for token in tokens:
        text = token.strip()
        if not text:
            # A stray comma like '1,,2' or '1,' -> malformed input.
            raise ValueError(f"empty range component in {token!r}")

        if '-' in text:
            start_str, _, end_str = text.partition('-')
            try:
                start = int(start_str.strip())
                end = int(end_str.strip())
            except ValueError:
                raise ValueError(f"malformed range {text!r}") from None
            if start > end:
                raise ValueError(f"reversed range {text!r}")
            numbers.extend(range(start, end + 1))
        else:
            try:
                numbers.append(int(text))
            except ValueError:
                raise ValueError(f"malformed number {text!r}") from None

    return numbers
```

A couple of sensible extra behaviors worth noting:
- A stray/double comma like `'1,,2'` or trailing `'1-3,'` raises a clear `ValueError` (malformed input) rather than silently producing garbage — only fully empty or whitespace-only *input* is treated leniently.
- Non-numeric fragments like `'1-3-5'` produce `ValueError: malformed range '1-3-5'` instead of a raw `int()` traceback.
