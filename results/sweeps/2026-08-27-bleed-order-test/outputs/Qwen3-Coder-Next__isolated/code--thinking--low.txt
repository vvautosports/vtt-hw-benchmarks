```python
def parse_ranges(s: str) -> list[int]:
    """
    Parse a string like '1-3,5,7-9' into [1, 2, 3, 5, 7, 8, 9].
    
    Requirements:
    - Ignore whitespace around numbers and commas.
    - A reversed range like '9-7' raises ValueError.
    - Empty or whitespace-only input returns [].
    
    Args:
        s (str): The string to parse.
        
    Returns:
        list[int]: A list of integers parsed from the input string.
        
    Raises:
        ValueError: If a reversed range is found (start > end).
        
    Examples:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]
        
        >>> parse_ranges('  1 - 3 , 5 , 7 - 9  ')
        [1, 2, 3, 5, 7, 8, 9]
        
        >>> parse_ranges('')
        []
        
        >>> parse_ranges('   ')
        []
        
        >>> parse_ranges('10-12,15,20-22')
        [10, 11, 12, 15, 20, 21, 22]
        
        >>> parse_ranges('9-7')
        Traceback (most recent call last):
            ...
        ValueError: Reversed range: 9-7 (start > end)
    """
    if not s or s.strip() == '':
        return []
    
    result = []
    parts = s.split(',')
    
    for part in parts:
        part = part.strip()
        if '-' in part:
            range_parts = part.split('-')
            if len(range_parts) != 2:
                raise ValueError(f"Invalid range format: {part}")
            start_str, end_str = range_parts
            start_str = start_str.strip()
            end_str = end_str.strip()
            
            if not start_str or not end_str:
                raise ValueError(f"Invalid range format: {part}")
            
            start = int(start_str)
            end = int(end_str)
            
            if start > end:
                raise ValueError(f"Reversed range: {part} (start > end)")
            
            result.extend(range(start, end + 1))
        else:
            if part:
                result.append(int(part))
    
    return result
```
