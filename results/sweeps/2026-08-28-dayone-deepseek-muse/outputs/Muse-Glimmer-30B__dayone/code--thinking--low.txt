```python
def parse_ranges(s: str) -> list[int]:
    """
    Parse a comma separated list of numbers and ranges.

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 - 3 , 5 , 7 - 9 ')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('5')
    [5]
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError
    """
    if not s or not s.strip():
        return []

    result: list[int] = []
    for part in s.split(','):
        part = part.strip()
        if not part:
            continue

        if '-' in part:
            left_str, sep, right_str = part.partition('-')
            if not sep or not left_str.strip() or not right_str.strip():
                raise ValueError(f"Invalid range: {part}")
            start = int(left_str.strip())
            end = int(right_str.strip())
            if start > end:
                raise ValueError(f"Reversed range: {start}-{end}")
            result.extend(range(start, end + 1))
        else:
            result.append(int(part))

    return result
```
