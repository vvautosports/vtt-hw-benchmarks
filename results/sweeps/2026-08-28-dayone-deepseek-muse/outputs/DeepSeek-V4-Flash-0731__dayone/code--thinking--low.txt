def parse_ranges(s: str) -> list[int]:
    """
    Parse a string of comma-separated numbers and ranges into a flat list of ints.

    The string may contain whitespace around numbers and commas. A range is
    written as 'a-b' and expands to all integers from a to b inclusive.
    If a reversed range (start > end) is encountered, ValueError is raised.
    Empty or whitespace-only input returns an empty list.

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 - 3 , 5 , 7 - 9 ')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('1,2,3')
    [1, 2, 3]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
    ValueError: Reversed range: 9 > 7
    >>> parse_ranges('1-3,10-12')
    [1, 2, 3, 10, 11, 12]
    """
    # Handle empty or whitespace-only input
    if s.strip() == '':
        return []

    result = []
    # Split by commas and strip whitespace from each token
    for token in s.split(','):
        token = token.strip()
        if not token:  # token empty (e.g., "1,,2") → invalid
            raise ValueError(f"Empty token in input: {s!r}")

        # Check if token is a range (contains '-')
        if '-' in token:
            # Split into start and end parts
            parts = token.split('-')
            if len(parts) != 2:
                raise ValueError(f"Invalid range format: {token!r}")
            start_str, end_str = parts[0].strip(), parts[1].strip()
            if not start_str or not end_str:
                raise ValueError(f"Invalid range format: {token!r}")
            start, end = int(start_str), int(end_str)
            if start > end:
                raise ValueError(f"Reversed range: {start} > {end}")
            # Extend result with the range
            result.extend(range(start, end + 1))
        else:
            # Single number
            result.append(int(token))

    return result
