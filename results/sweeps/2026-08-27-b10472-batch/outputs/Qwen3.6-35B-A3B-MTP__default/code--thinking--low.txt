```python
def parse_ranges(s: str) -> list[int]:
    """Parse a comma-separated string of numbers and ranges into a flat list.

    Examples:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]

        >>> parse_ranges('  1 , 2-4  , 6 ')
        [1, 2, 3, 4, 6]

        >>> parse_ranges('')
        []

        >>> parse_ranges('   ')
        []

        >>> parse_ranges('9-7')
        Traceback (most recent call last):
            ...
        ValueError: reversed range '9-7'

        >>> parse_ranges('5')
        [5]
    """
    if not s or s.strip() == '':
        return []

    result = []
    parts = s.split(',')

    for part in parts:
        part = part.strip()
        if not part:
            continue  # handle trailing commas gracefully

        if '-' in part:
            start_str, end_str = part.split('-', 1)
            start = int(start_str.strip())
            end = int(end_str.strip())
            if start > end:
                raise ValueError(f"reversed range '{part}'")
            result.extend(range(start, end + 1))
        else:
            result.append(int(part))

    return result
```

**How it works:**

1. **Empty/whitespace input** → returns `[]` early.
2. **Split on commas**, then strip whitespace from each part (handles `' 1 , 2-4  '` gracefully).
3. If a part contains `-`, split into start/end integers and raise `ValueError` if `start > end`. Otherwise, extend the result with `range(start, end + 1)`.
4. If no `-`, parse as a single integer.
