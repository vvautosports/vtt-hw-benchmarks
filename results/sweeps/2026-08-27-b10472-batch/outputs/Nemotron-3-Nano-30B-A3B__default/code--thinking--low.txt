def parse_ranges(s: str) -> list[int]:
    """
    Parse a string containing comma‑separated ranges into a flat list of integers.

    The input may contain arbitrary whitespace around numbers and commas.
    A range is expressed as ``a-b`` (inclusive) and must satisfy ``a <= b``;
    otherwise a ``ValueError`` is raised.  An empty or whitespace‑only string
    yields an empty list.

    Doctests
    --------
    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 - 3 , 5 , 7 - 9 ')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('5')
    [5]
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError: invalid range '9-7': start (9) > end (7)
    """
    if not s or s.strip() == "":
        return []

    result: list[int] = []
    for part in s.split(','):
        token = part.strip()
        if token == "":
            continue  # skip empty tokens that may arise from extra commas

        if '-' in token:
            # split on the first hyphen only; multiple hyphens are not supported
            start_str, end_str = token.split('-', 1)
            try:
                start = int(start_str.strip())
                end = int(end_str.strip())
            except ValueError as exc:
                raise ValueError(f"invalid range '{token}': "
                                 f"start ({start_str.strip()}) or end ({end_str.strip()}) "
                                 f"is not an integer") from exc

            if start > end:
                raise ValueError(f"invalid range '{token}': start ({start}) > end ({end})")
            result.extend(range(start, end + 1))
        else:
            # single number
            result.append(int(token))

    return result
