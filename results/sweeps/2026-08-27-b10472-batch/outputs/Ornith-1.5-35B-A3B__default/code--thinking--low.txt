```python
def parse_ranges(s: str) -> list[int]:
    """Parse a comma-separated string of numbers and inclusive ranges.

    Whitespace around numbers and commas is ignored. A reversed range
    (e.g. '9-7') raises ``ValueError``. Empty or whitespace-only input
    returns an empty list.

    Examples
    --------
    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges(' 2 , 4-6 , 8 ')
    [2, 4, 5, 6, 8]
    >>> parse_ranges('5-3')
    Traceback (most recent call last):
        ...
    ValueError: reversed range '5-3'
    >>> parse_ranges('1-3,1-3')
    [1, 2, 3, 1, 2, 3]
    """
    result: list[int] = []
    token = s.strip()
    if not token:
        return result

    for part in token.split(','):
        part = part.strip()
        if not part:
            raise ValueError(f"empty range segment in {part!r}")
        if '-' in part:
            start_str, _, end_str = part.partition('-')
            if '-' in end_str or not start_str.strip() or not end_str.strip():
                raise ValueError(f"malformed range segment {part!r}")
            start, end = int(start_str), int(end_str)
            if end < start:
                raise ValueError(f"reversed range {part!r}")
            result.extend(range(start, end + 1))
        else:
            result.append(int(part))
    return result
```
