```python
def parse_ranges(s: str) -> list[int]:
    """
    Parse a string of ranges into a list of integers.
    
    Args:
        s: A string like '1-3,5,7-9'
        
    Returns:
        A list of integers from the ranges
        
    Raises:
        ValueError: If a range is reversed (like '9-7')
        
    Examples:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]
        >>> parse_ranges('  1 - 3 , 5 , 7 - 9  ')
        [1, 2, 3, 5, 7, 8, 9]
        >>> parse_ranges('10-12,15,17-19')
        [10, 11, 12, 15, 17, 18, 19]
        >>> parse_ranges('5')
        [5]
        >>> parse_ranges('')
        []
        >>> parse_ranges('9-7')
        Traceback (most recent call last):
            ...
        ValueError: Reversed range: 9-7
    """
    # Handle empty or whitespace-only input
    if not s or not s.strip():
        return []
    
    result = []
    
    # Split by comma and process each part
    for part in s.split(','):
        part = part.strip()
        if not part:  # Skip empty parts
            continue
            
        # Check if it's a range (contains '-')
        if '-' in part:
            start_str, end_str = part.split('-')
            start = int(start_str.strip())
            end = int(end_str.strip())
            
            # Check for reversed range
            if start > end:
                raise ValueError(f"Reversed range: {start}-{end}")
            
            # Add the range to result
            result.extend(range(start, end + 1))
        else:
            # Single number
            result.append(int(part.strip()))
    
    return result
```
