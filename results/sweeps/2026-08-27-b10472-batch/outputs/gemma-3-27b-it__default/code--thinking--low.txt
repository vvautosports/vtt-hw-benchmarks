def parse_ranges(s: str) -> list[int]:
    """Parses a string like '1-3,5,7-9' into a list of integers.

    Args:
        s: The string to parse.

    Returns:
        A list of integers representing the ranges.

    Raises:
        ValueError: If a reversed range (e.g., '9-7') is encountered.

    Examples:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]
        >>> parse_ranges('1, 2, 3')
        [1, 2, 3]
        >>> parse_ranges('5-7')
        [5, 6, 7]
        >>> parse_ranges('  1 - 3 , 5,7-9  ')
        [1, 2, 3, 5, 7, 8, 9]
        >>> parse_ranges('')
        []
        >>> parse_ranges('9-7')
        Traceback (most recent call last):
        ValueError: Reversed range: 9-7
        >>> parse_ranges('   ')
        []
    """
    result = []
    if not s.strip():
        return result

    for part in s.split(','):
        part = part.strip()
        if '-' in part:
            try:
                start, end = map(int, part.split('-'))
                if start > end:
                    raise ValueError(f"Reversed range: {part}")
                result.extend(range(start, end + 1))
            except ValueError:
                raise
        else:
            try:
                result.append(int(part))
            except ValueError:
                raise ValueError(f"Invalid input: {part}")
    return result
