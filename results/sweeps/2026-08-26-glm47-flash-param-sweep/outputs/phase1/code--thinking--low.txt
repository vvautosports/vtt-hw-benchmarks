def parse_ranges(s: str) -> list[int]:
    """
    Parse a string containing numeric ranges into a list of integers.
    
    Examples:
    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 - 3 , 5 , 7 - 9 ')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError: Invalid range: 9-7 (end before start)
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('2-5,10')
    [2, 3, 4, 5, 10]
    """
    if not s or s.strip() == '':
        return []
    
    result = []
    # Split by commas, ignore empty parts
    parts = [p.strip() for p in s.split(',')]
    
    for part in parts:
        if '-' in part:
            # Handle ranges
            start_str, end_str = [x.strip() for x in part.split('-')]
            try:
                start = int(start_str)
                end = int(end_str)
            except ValueError as e:
                raise ValueError(f"Invalid number in range: {part}") from e
            
            if end < start:
                raise ValueError(f"Invalid range: {part} (end before start)")
            
            result.extend(range(start, end + 1))
        else:
            # Handle single numbers
            try:
                num = int(part)
            except ValueError as e:
                raise ValueError(f"Invalid number: {part}") from e
            result.append(num)
    
    return result
