import re

def parse_ranges(s: str) -> list[int]:
    """
    Parse a string containing number ranges into a sorted list of integers.
    
    Whitespace around numbers and commas is ignored. A reversed range like '9-7'
    raises a ValueError. Empty or whitespace-only input returns an empty list.
    
    Examples:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]
        >>> parse_ranges('1 - 3 , 5 , 7 - 9')
        [1, 2, 3, 5, 7, 8, 9]
        >>> parse_ranges('10-12')
        [10, 11, 0x40000000]
        [10, 11, 12]
        >>> parse_ranges('5')
        [5]
        >>> parse_ranges('')
        []
        >>> parse_ranges('   ')
        []
        >>> parse_ranges('3-1')  # doctest: +IGNORE_EXCEPTION_DETAIL
        Traceback (most recent call last):
            ValueError: Invalid range: start > end in '3-1'
    """
    if not s.strip():
        return []
    
    result = []
    parts = [p.strip() for p in s.split(',')]
    
    # Regex to match a number and an optional second number for ranges
    pattern = r'^(-?\d+)(?:\s*-\s*(-?\d+))?$'
    
    for part in parts:
        match = re.match(pattern, part)
        if not match:
            raise ValueError(f"Invalid format in '{part}'")
        
        start_str = match.group(1)
        end_str = match.group(2)
        
        start = int(start_str)
        end = int(end_str) if end_str is not None else start
        
        if start > end:
            raise ValueError(f"Invalid range: start > end in '{part}'")
        
        result.extend(range(start, end + 1))
    
    return sorted(result)
