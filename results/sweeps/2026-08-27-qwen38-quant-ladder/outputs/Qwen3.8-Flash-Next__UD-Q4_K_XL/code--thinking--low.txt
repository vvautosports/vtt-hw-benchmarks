def parse_ranges(s: str) -> list[int]:
    """Parse a range string into a flat list of integers.

    Accepts comma-separated tokens where each token is either a single integer
    or an inclusive range 'start-end'. Whitespace around numbers and commas is
    ignored. Reversed ranges (where start > end) raise ValueError.

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 4 , 6-8 ')
    [4, 6, 7, 8]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('5')
    [5]
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError: reversed range: 9-7
    >>> parse_ranges('3 - 1')
    Traceback (most recent call last):
        ...
    ValueError: reversed range: 3 - 1
    """
    s = s.strip()
    if not s:
        return []

    result: list[int] = []
    for part in s.split(','):
        part = part.strip()
        if '-' in part:
            left, right = part.split('-', 1)
            start = int(left.strip())
            end = int(right.strip())
            if start > end:
                raise ValueError(f"reversed range: {part}")
            result.extend(range(start, end + 1))
        else:
            result.append(int(part))

    return result

**How it works:**

| Step | Detail |
|------|--------|
| **Empty check** | Strip the whole string; if nothing remains, return `[]` immediately. |
| **Tokenize** | Split on `,`, then strip each token. |
| **Range vs single** | If a token contains `-`, split on the first `-`, parse both sides as integers, and expand with `range(start, end+1)`. Otherwise parse the token as a single integer. |
| **Reversed guard** | If `start > end`, raise `ValueError` with a descriptive message. |
| **Whitespace tolerance** | Every boundary (outer string, around commas, around `-`) is handled by `.strip()`. |
