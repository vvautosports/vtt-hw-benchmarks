def parse_ranges(s: str) -> list[int]:
    """
    Parse a string like '1-3,5,7-9' into [1, 2, 3, 5, 7, 8, 9].

    Requirements:
    - Ignore whitespace around numbers and commas
    - A reversed range like '9-7' raises ValueError
    - Empty or whitespace-only input returns []

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 , 2 , 3 ')
    [1, 2, 3]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
    ...
    ValueError: Reversed range '9-7' is not allowed
    >>> parse_ranges('5-3,10')
    Traceback (most recent call last):
    ...
    ValueError: Reversed range '5-3' is not allowed
    """
    # Strip the input and check if it's empty or whitespace-only
    s = s.strip()
    if not s:
        return []
    
    result = []
    # Split by commas, stripping whitespace from each part
    parts = [part.strip() for part in s.split(',')]
    
    for part in parts:
        if not part:
            continue  # skip empty parts from things like "1,,2"
        
        # Check if it's a range (contains '-')
        if '-' in part:
            # Split by '-' and strip each side
            sides = [side.strip() for side in part.split('-')]
            if len(sides) != 2:
                raise ValueError(f"Invalid range format: '{part}'")
            try:
                start = int(sides[0])
                end = int(sides[1])
            except ValueError:
                raise ValueError(f"Invalid number in range: '{part}'")
            
            if start > end:
                raise ValueError(f"Reversed range '{part}' is not allowed")
            
            # Add all numbers from start to end inclusive
            result.extend(range(start, end + 1))
        else:
            # Single number
            try:
                num = int(part)
            except ValueError:
                raise ValueError(f"Invalid number: '{part}'")
            result.append(num)
    
    return result
