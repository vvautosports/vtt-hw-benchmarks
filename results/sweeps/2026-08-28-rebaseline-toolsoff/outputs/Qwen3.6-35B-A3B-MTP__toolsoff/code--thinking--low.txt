def parse_ranges(s: str) -> list[int]:
    """
    Parses a string of comma-separated ranges and numbers into a flat list of integers.
    
    Whitespace around numbers and commas is ignored.
    Reversed ranges (e.g., '9-7') raise a ValueError.
    Empty or whitespace-only strings return an empty list.
    
    Examples:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]
        
        >>> parse_ranges(' 10 , 12-15 , 20 ')
        [10, 12, 13, 14, 15, 20]
        
        >>> parse_ranges('42')
        [42]
        
        >>> parse_ranges('')
        []
        
        >>> parse_ranges('   ')
        []
        
        >>> parse_ranges('9-7')
        Traceback (most recent call last):
            ...
        ValueError: Reversed range detected: 9-7
    """
    if not s.strip():
        return []

    result = []
    for part in s.split(','):
        part = part.strip()
        if not part:
            continue  # Skip empty segments from consecutive commas or trailing separators

        if '-' in part:
            start_str, end_str = part.split('-', 1)
            try:
                start, end = int(start_str), int(end_str)
            except ValueError:
                raise ValueError(f"Invalid number format in segment: '{part}'")
            
            if start > end:
                raise ValueError(f"Reversed range detected: {start}-{end}")
                
            result.extend(range(start, end + 1))
        else:
            try:
                result.append(int(part))
            except ValueError:
                raise ValueError(f"Invalid number format in segment: '{part}'")

    return result
