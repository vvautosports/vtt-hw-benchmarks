    s = s.strip()

    result = []
    parts = s.split(',')

            right = right.strip()

                end = int(right)

        else:

                num = int(part)
                result.append(num)
            except ValueError:

def parse_ranges(s: str) -> list[int]:
    """
    Parse a string of comma-separated ranges into a list of integers.

    A range is specified as 'start-end' inclusive. Whitespace around numbers,
    commas, and dashes is ignored. A reversed range (e.g., '9-7') raises ValueError.
    Empty or whitespace-only input returns an empty list.

    Examples:
    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1-3, 5 ,7-9 ')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError
    >>> parse_ranges('')
    []
    >>> parse_ranges('42')
    [42]
    """
    s = s.strip()
    if not s:
        return []

    result = []
    parts = s.split(',')
    for part in parts:
        part = part.strip()
        if not part:
            continue  # skip empty segments (e.g., trailing/extra commas)
        if '-' in part:
            # Split only on the first dash to allow spaces around it
            left, right = part.split('-', 1)
            left = left.strip()
            right = right.strip()
            if not left or not right:
                raise ValueError(f"Invalid range: {part}")
            try:
                start = int(left)
                end = int(right)
            except ValueError:
                raise ValueError(f"Invalid number in range: {part}")
            if start > end:
                raise ValueError
            result.extend(range(start, end + 1))
        else:
            try:
                num = int(part)
                result.append(num)
            except ValueError:
                raise ValueError(f"Invalid number: {part}")
    return result

def parse_ranges(s: str) -> list[int]:
    """
    Parse a string of comma-separated ranges into a list of integers.

    A range is specified as 'start-end' inclusive. Whitespace around numbers,
    commas, and dashes is ignored. A reversed range (e.g., '9-7') raises ValueError.
    Empty or whitespace-only input returns an empty list.

    Examples:
    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1-3, 5 ,7-9 ')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('42')
    [42]
    """
    s = s.strip()
    if not s:
        return []

    result = []
    parts = s.split(',')
    for part in parts:
        part = part.strip()
        if not part:
            continue  # skip empty segments (e.g., trailing/extra commas)

        if '-' in part:
            # Split only on the first dash to allow spaces around it
            left, right = part.split('-', 1)
            left = left.strip()
            right = right.strip()
            if not left or not right:
                raise ValueError(f"Invalid range: {part}")
            try:
                start = int(left)
                end = int(right)
            except ValueError:
                raise ValueError(f"Invalid number in range: {part}")
            if start > end:
                raise ValueError  # reversed range
            result.extend(range(start, end + 1))
        else:
            try:
                num = int(part)
                result.append(num)
            except ValueError:
                raise ValueError(f"Invalid number: {part}")

    return result
