def parse_ranges(s: str) -> list[int]:
    """
    Convert a comma‑separated list of numbers and numeric ranges into a flat list
    of integers.

    The syntax accepted is a sequence of *items* separated by commas.
    An item can be either

    * a single integer, e.g. ``5``
    * a range expressed as ``start-end`` where ``start`` and ``end`` are integers
      and ``start <= end`` (the range is inclusive).

    Whitespace around numbers, dashes and commas is ignored.
    If a range is written in reverse order (``9-7``) a :class:`ValueError` is raised.
    An empty string or a string that contains only whitespace yields an empty list.

    Parameters
    ----------
    s: str
        The textual representation of the ranges.

    Returns
    -------
    list[int]
        A list of the individual integers in the order they appear in the
        input string.

    Raises
    ------
    ValueError
        If a token cannot be interpreted as an integer or a range, or if a
        range is reversed (``start > end``).

    Doctests
    --------
    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]

    >>> parse_ranges(' 10 , 12-14 , 16 ')
    [10, 12, 13, 14, 16]

    >>> parse_ranges('42')
    [42]

    >>> parse_ranges('')      # empty input
    []

    >>> parse_ranges('   ')   # whitespace‑only input
    []

    >>> parse_ranges('3-3,5')   # a range where start == end is allowed
    [3, 5]

    >>> parse_ranges('9-7')      # reversed range -> error
    Traceback (most recent call last):
        ...
    ValueError: reversed range '9-7' (start > end)

    >>> parse_ranges('1-2, a, 5')   # non‑numeric token -> error
    Traceback (most recent call last):
        ...
    ValueError: invalid token 'a'
    """
    # Strip outer whitespace; early‑exit for empty input.
    stripped = s.strip()
    if not stripped:
        return []

    result: list[int] = []
    # Split on commas – commas may be surrounded by whitespace.
    for raw_item in stripped.split(','):
        item = raw_item.strip()
        if not item:
            # ``',,'`` or a trailing comma would produce an empty token.
            raise ValueError("empty token between commas")
        if '-' in item:
            # Range token
            parts = item.split('-')
            if len(parts) != 2:
                raise ValueError(f"invalid range token '{item}'")
            start_str, end_str = parts[0].strip(), parts[1].strip()
            try:
                start = int(start_str)
                end = int(end_str)
            except ValueError as exc:
                raise ValueError(f"invalid token '{item}'") from exc
            if start > end:
                raise ValueError(f"reversed range '{item}' (start > end)")
            # Extend the result with the inclusive range.
            result.extend(range(start, end + 1))
        else:
            # Single number token
            try:
                num = int(item)
            except ValueError as exc:
                raise ValueError(f"invalid token '{item}'") from exc
            result.append(num)

    return result
