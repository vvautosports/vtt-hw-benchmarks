def parse_ranges(s: str) -> list[int]:
    """
    Parses a string of comma-separated numbers and ranges into a list of integers.
    
    Examples:
    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 , 2-4 ')
    [1, 2, 3, 4]
    >>> parse_ranges('  ')
    []
    >>> parse_ranges('')
    []
    >>> parse_ranges('5')
    [5]
    >>> parse_ranges('10-8')
    Traceback (most recent call last):
        ...
    ValueError: Reversed range: 10-8
    """
    # Return empty list for empty or whitespace-only input
    if not s or not s.strip():
        return []

    result = []
    # Split by comma to separate individual numbers or ranges
    parts = s.split(',')

    for part in parts:
        part = part.strip()
        if not part:
            continue  # Skip empty segments if there are trailing/double commas
            
        if '-' in part:
            # Split the range into start and end
            try:
                start_str, end_str = part.split('-')
                start, end = int(start_str), int(end_str)
            except ValueError:
                raise ValueError(f"Invalid range format: {part}")

            # Check for reversed range
            if start > end:
                raise ValueError(f"Reversed range: {part}")
            
            # Add all integers in the range (inclusive)
            result.extend(range(start, end + 1))
        else:
            # Parse as a single integer
            result.append(int(part))

    return result
