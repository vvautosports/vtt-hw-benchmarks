def parse_ranges(s: str) -> list[int]:
    """
    Parses a string containing single numbers or ranges (e.g., '1-3,5,7-9') 
    into a flat list of integers.

    Args:
        s: A string containing comma-separated numbers and ranges.

    Returns:
        A list of integers.

    Raises:
        ValueError: If a range is reversed (e.g., '9-7').

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 - 3 , 5 , 7 - 9 ')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('1, 2, 3')
    [1, 2, 3]
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('')
    []
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError: Range start (9) cannot be greater than range end (7)
    """
    # Return empty list for empty or whitespace-only strings
    if not s or not s.strip():
        return []

    result = []
    # Split by comma to get individual components (numbers or ranges)
    parts = s.split(',')

    for part in parts:
        part = part.strip()
        if not part:
            continue
            
        if '-' in part:
            # Handle range logic
            try:
                start_str, end_str = part.split('-')
                start, end = int(start_str.strip()), int(end_str.strip())
            except ValueError as e:
                # Re-raise as ValueError if the split doesn't result in exactly two ints
                raise ValueError(f"Invalid range format: '{part}'") from e
            
            if start > end:
                raise ValueError(f"Range start ({start}) cannot be greater than range end ({end})")
            
            # Use end + 1 because range() is exclusive at the stop parameter
            result.extend(range(start, end + 1))
        else:
            # Handle single number logic
            try:
                result.append(int(part))
            except ValueError as e:
                raise ValueError(f"Invalid number format: '{part}'") from e

    return result
