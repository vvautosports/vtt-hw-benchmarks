def parse_ranges(s: str) -> list[int]:
    """
    Parses a string of comma-separated numbers and ranges into a list of integers.
    Example input: '1-3,5,7-9' -> [1, 2, 3, 5, 7, 8, 9]

    Requirements:
    - Ignore whitespace around numbers and commas.
    - A reversed range like '9-7' raises ValueError.
    - Empty or whitespace-only input returns [].

    >>> parse_ranges('1-3,5,7-9')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges(' 1 - 3 , 5 , 7 - 9 ')
    [1, 2, 3, 5, 7, 8, 9]
    >>> parse_ranges('')
    []
    >>> parse_ranges('   ')
    []
    >>> parse_ranges('9-7')
    Traceback (most recent call last):
        ...
    ValueError: Reversed range: 9-7
    """
    # Handle empty or whitespace-only input
    s = s.strip()
    if not s:
        return []

    result = []
    # Split by comma to isolate individual numbers or ranges
    parts = s.split(',')

    for part in parts:
        part = part.strip()
        if not part:
            continue

        if '-' in part:
            # Handle range format 'start-end'
            bounds = part.split('-')
            
            # Ensure there are exactly two parts for a valid range
            if len(bounds) != 2:
                raise ValueError(f"Invalid range format: '{part}'")

            try:
                start = int(bounds[0].strip())
                end = int(bounds[1].strip())
            except ValueError:
                raise ValueError(f"Non-integer found in range: '{part}'")

            # Check for reversed range requirement
            if start > end:
                raise ValueError(f"Reversed range: {start}-{end}")

            # Extend result with the range of integers (inclusive)
            result.extend(range(start, end + 1))
        else:
            # Handle single number format
            try:
                result.append(int(part))
            except ValueError:
                raise ValueError(f"Non-integer found: '{part}'")

    return result
