```python
def parse_ranges(s: str) -> list[int]:
    """Parse range string like '1-3,5,7-9' into list of ints.

    Args:
        s (str): Input string with comma-separated values and ranges.

    Returns:
        list[int]: List of integers.

    Raises:
        ValueError: If a reversed range is given (e.g., '9-7').

    Examples:
        >>> parse_ranges('1-3,5,7-9')
        [1, 2, 3, 5, 7, 8, 9]
        >>> parse_ranges('1,2,3')
        [1, 2, 3]
        >>> parse_ranges('-1-0')
        ValueError: Range must be non-reversed (e.g., "9-7" is invalid)
        >>> parse_ranges(' ')
        []
        >>> parse_ranges('a-b,c-d')
        []
    """
    # Handle empty or whitespace-only input
    if not s or s.strip() == '':
        return []

    import re

    result = []
    for part in s.split(','):
        part = part.strip()  # ignore whitespace around commas
        if part == '':
            continue

        # Try to match a range pattern (with optional spaces)
        m = re.match(r'^(\d+)\s*-\s*(\d+)$', part)
        if not m:
            # Not a range, treat as single number
            n = int(part)
            result.append(n)
        else:
            start, end = map(int, (m.group(1), m.group(2)))
            if start > end:
                raise ValueError('Range must be non-reversed (e.g., "9-7" is invalid)')
            for i in range(start, end + 1):
                result.append(i)

    return result
```
